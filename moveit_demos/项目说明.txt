🚀 快速开始（复制粘贴即可）
--------------------------------------------------------------------------------

# 步骤 1: 进入目录并安装依赖
cd ~/agx_arm_ws/src/agx_arm_ros/moveit_demos
pip3 install -r requirements.txt
chmod +x *.py *.sh

# 步骤 2: 启动机械臂（选择一种方式）

## 方式 A：交互式启动（推荐）
./quick_start.sh

## 方式 B：手动启动（真机 + 夹爪）
ros2 launch agx_arm_ctrl start_single_agx_arm_moveit.launch.py \
  can_port:=can0 arm_type:=piper effector_type:=agx_gripper

## 方式 C：仅仿真
ros2 launch agx_arm_moveit demo.launch.py arm_type:=piper

# 步骤 3: 新终端运行 Demo
cd ~/agx_arm_ws/src/agx_arm_ros/moveit_demos
python3 moveit_simple_demo.py

📞 常用调试命令
--------------------------------------------------------------------------------
# 检查机械臂状态
ros2 topic echo /feedback/arm_status

# 使能机械臂
ros2 service call /enable_agx_arm std_srvs/srv/SetBool "{data: true}"

# 打开控制门
ros2 service call /control_enable std_srvs/srv/SetBool "{data: true}"

# 回零位
ros2 service call /move_home std_srvs/srv/Empty

# 急停
ros2 service call /emergency_stop std_srvs/srv/Empty
